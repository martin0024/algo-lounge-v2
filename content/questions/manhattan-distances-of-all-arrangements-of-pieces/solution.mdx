## Reference solution

```python
MOD = 10**9 + 7


def distance_sum(m: int, n: int, k: int) -> int:
    # Every pair of cells is counted once per arrangement that places pieces on
    # both of them: C(mn - 2, k - 2) arrangements. Sum the pair distances by
    # row gap d (m - d row pairs, n * n cell pairs each) and column gap.
    cells = m * n
    rows = sum(d * (m - d) for d in range(1, m)) * n * n
    cols = sum(d * (n - d) for d in range(1, n)) * m * m

    fact = [1] * (cells + 1)
    for i in range(1, cells + 1):
        fact[i] = fact[i - 1] * i % MOD
    def comb(a, b):
        if b < 0 or b > a:
            return 0
        return fact[a] * pow(fact[b], MOD - 2, MOD) % MOD * pow(fact[a - b], MOD - 2, MOD) % MOD

    return (rows + cols) % MOD * comb(cells - 2, k - 2) % MOD
```
